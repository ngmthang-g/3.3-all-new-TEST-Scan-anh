cmake_minimum_required(VERSION 3.24)
project(ThanLongItemConsolidator VERSION 4.6 LANGUAGES CXX RC)

if(NOT WIN32)
    message(FATAL_ERROR "ThanLongItemConsolidator is Windows-only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "ThanLongItemConsolidator must be built x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(ThanLongCleanRouteBridge SHARED src/bridge.cpp)
target_include_directories(ThanLongCleanRouteBridge PRIVATE src)
target_compile_definitions(ThanLongCleanRouteBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ThanLongCleanRouteBridge PRIVATE user32 kernel32)

add_executable(ThanLongItemConsolidator WIN32 src/controller.cpp src/telegram_notifier.cpp src/telegram_account_filter.cpp resources/app.rc)
target_include_directories(ThanLongItemConsolidator PRIVATE src)
target_compile_definitions(ThanLongItemConsolidator PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ThanLongItemConsolidator PRIVATE comctl32 comdlg32 user32 kernel32 gdi32 winhttp crypt32)
target_link_options(ThanLongItemConsolidator PRIVATE /MANIFEST:NO)

add_executable(route_logic_tests src/route_logic_test.cpp)
target_include_directories(route_logic_tests PRIVATE src)
add_executable(rotation_logic_tests src/rotation_logic_test.cpp)
target_include_directories(rotation_logic_tests PRIVATE src)
add_executable(trade_coordinator_logic_tests src/trade_coordinator_logic_test.cpp)
target_include_directories(trade_coordinator_logic_tests PRIVATE src)
add_executable(background_ui_logic_tests src/background_ui_logic_test.cpp)
target_include_directories(background_ui_logic_tests PRIVATE src)
add_executable(fixed_slot_sell_logic_tests src/fixed_slot_sell_logic_test.cpp)
target_include_directories(fixed_slot_sell_logic_tests PRIVATE src)
add_executable(unity_geometry_logic_tests src/unity_geometry_logic_test.cpp)
target_include_directories(unity_geometry_logic_tests PRIVATE src)
add_executable(internal_ui_click_logic_tests src/internal_ui_click_logic_test.cpp)
target_include_directories(internal_ui_click_logic_tests PRIVATE src)
add_executable(travel_fight_guard_logic_tests src/travel_fight_guard_logic_test.cpp)
target_include_directories(travel_fight_guard_logic_tests PRIVATE src)
add_executable(auto_fight_retry_logic_tests src/auto_fight_retry_logic_test.cpp)
target_include_directories(auto_fight_retry_logic_tests PRIVATE src)

add_executable(inventory_filter_logic_tests src/inventory_filter_logic_test.cpp)
target_include_directories(inventory_filter_logic_tests PRIVATE src)
add_executable(dungeon_logic_tests src/dungeon_logic_test.cpp)
target_include_directories(dungeon_logic_tests PRIVATE src)
add_executable(dungeon_v45_logic_tests src/dungeon_v45_logic_test.cpp)
target_include_directories(dungeon_v45_logic_tests PRIVATE src)
add_executable(travel_network_logic_tests src/travel_network_logic_test.cpp)

if(MSVC)
    target_compile_options(ThanLongCleanRouteBridge PRIVATE /utf-8 /W4 /permissive- /FIstdlib.h)
    foreach(t ThanLongItemConsolidator route_logic_tests rotation_logic_tests trade_coordinator_logic_tests background_ui_logic_tests fixed_slot_sell_logic_tests unity_geometry_logic_tests internal_ui_click_logic_tests travel_fight_guard_logic_tests auto_fight_retry_logic_tests auto_pk_logic_tests telegram_logic_tests inventory_filter_logic_tests dungeon_logic_tests dungeon_v45_logic_tests travel_network_logic_tests)
